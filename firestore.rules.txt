rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    match /users/{userId} {
      // The user doc holds the `subscription` map ({active, currentPeriodEnd,
      // ...}) written only by the Paystack webhook via the Admin SDK, which
      // bypasses these rules entirely. Clients get read-only access so the
      // app can show subscription status — never write access, so the
      // $2/month gate can't be flipped from the browser.
      allow get: if request.auth != null && request.auth.uid == userId;
      allow list, write: if false;

      match /reminders/{reminderId} {
        allow read, delete: if request.auth != null && request.auth.uid == userId;
        // Reminders are a premium feature: creating/updating one requires
        // both ownership AND an active subscription on the parent user doc.
        // "sent" is reset to false by the client on create but only the
        // Cloud Function (Admin SDK, bypasses rules) should ever flip it
        // back to true.
        allow create, update: if request.auth != null
          && request.auth.uid == userId
          && request.resource.data.sent == false
          && get(/databases/$(database)/documents/users/$(userId)).data.subscription.active == true;
      }

      match /fcmTokens/{tokenId} {
        allow read, write, delete: if request.auth != null && request.auth.uid == userId;
      }
    }

    // Deny everything else by default.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
