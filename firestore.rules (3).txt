rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    match /users/{userId} {
      // The user doc holds the `adUnlock` map ({active, expiresAt,
      // lastGrantedAt}), written only by /api/ads/unlock.js via the Admin
      // SDK, which bypasses these rules entirely. Clients get read-only
      // access so the app can show unlock status — never write access, so
      // the ad-unlock window can't be forged or extended from the browser.
      allow get: if request.auth != null && request.auth.uid == userId;
      allow list, write: if false;

      match /reminders/{reminderId} {
        allow read, delete: if request.auth != null && request.auth.uid == userId;
        // Reminders are gated at creation time only: creating one requires
        // both ownership AND an active ad-unlock window on the parent user
        // doc at that moment. Once created, a reminder is NOT re-gated at
        // delivery/update time — the ad-unlock window is short-lived and a
        // reminder set today may be due next week, so re-checking at
        // delivery would silently break far-future reminders. "sent" is
        // reset to false by the client on create but only the Cloud
        // Function (Admin SDK, bypasses rules) should ever flip it back
        // to true.
        allow create: if request.auth != null
          && request.auth.uid == userId
          && request.resource.data.sent == false
          && get(/databases/$(database)/documents/users/$(userId)).data.adUnlock.active == true
          && get(/databases/$(database)/documents/users/$(userId)).data.adUnlock.expiresAt > request.time.toMillis();
        allow update: if request.auth != null && request.auth.uid == userId;
      }

      match /fcmTokens/{tokenId} {
        allow read, write, delete: if request.auth != null && request.auth.uid == userId;
      }
    }

    // Deny everything else by default.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
